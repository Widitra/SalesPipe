function doGet(e){return h(e)}
function doPost(e){return h(e)}
function h(e){
  var ss=SpreadsheetApp.getActive();
  var b=e.postData?JSON.parse(e.postData.contents):{};
  var a=b.action||e.parameter.action||'getAll';
  if(a==='getAll')return out({status:'success',projects:rd(ss,'Projects'),stakeholders:rd(ss,'Stakeholders'),activityLogs:rd(ss,'ActivityLogs')});
  if(a==='saveProject'){
    up(ss,'Projects',b.project);
    var s=ss.getSheetByName('Stakeholders');
    var keep=s.getDataRange().getValues().filter(function(r,i){return i===0||r[1]!==b.project.ID});
    s.clearContents();
    s.getRange(1,1,keep.length,keep[0].length).setValues(keep);
    (b.stakeholders||[]).forEach(function(x){up(ss,'Stakeholders',x)});
  }
  if(a==='saveActivity')up(ss,'ActivityLogs',b.activity);
  if(a==='deleteProject'){
    ['Projects','Stakeholders','ActivityLogs'].forEach(function(n){
      var sh=ss.getSheetByName(n),c=n==='Projects'?0:1,v=sh.getDataRange().getValues();
      for(var i=v.length-1;i>0;i--){if(v[i][c]===b.id)sh.deleteRow(i+1)}
    });
  }
  return out({status:'success'});
}
function rd(ss,n){
  var v=ss.getSheetByName(n).getDataRange().getValues();
  var hd=v[0];
  return v.slice(1).map(function(r){var o={};hd.forEach(function(k,i){o[k]=r[i]});return o});
}
function up(ss,n,rec){
  var s=ss.getSheetByName(n);
  var v=s.getDataRange().getValues();
  var hd=v[0],ix=hd.indexOf('ID'),row=-1;
  v.forEach(function(r,i){if(i>0&&r[ix]===rec.ID)row=i+1});
  var d=hd.map(function(k){return rec[k]!==undefined?rec[k]:''});
  if(row>0)s.getRange(row,1,1,d.length).setValues([d]);else s.appendRow(d);
}
function out(o){return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON)}
